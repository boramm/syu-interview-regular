📋 Google Sheets Apps Script 코드
var SHEET_ID = '본인의_시트_ID_입력'; function doPost(e) { var ss = SpreadsheetApp.openById(SHEET_ID); var sheet = ss.getSheets()[0]; var data = e.parameter || {}; sheet.appendRow([ new Date(), data.projectTitle||'' ,data.applicantNum||'', data.applicantName||'', data.department||'', data.evaluatorName||'', data.evalDate||'', data.i1||0,data.i2||0,data.i3||0,data.i4||0,data.i5||0,data.interviewTotal||0, data.e1||0,data.e2||0,data.e3||0,data.e4||0,data.e5||0,data.e6||0, data.expertiseTotal||0,data.bohunPoint||0,data.finalScore||0, data.grade||'',data.comment||'' ]); return ContentService .createTextOutput(JSON.stringify({result:'ok'})) .setMimeType(ContentService.MimeType.JSON); } function doGet(e) { var ss = SpreadsheetApp.openById(SHEET_ID); var sheet = ss.getSheets()[0]; var rows = sheet.getDataRange().getValues(); return ContentService .createTextOutput(JSON.stringify({status:'ok',rows:rows})) .setMimeType(ContentService.MimeType.JSON); }
① Google Sheets 생성 → ② 확장프로그램 › Apps Script → ③ 위 코드 붙여넣기 (SHEET_ID 교체) → ④ 배포 › 새 배포 › 웹앱 › 모든 사용자 → ⑤ URL 복사해서 위 입력란에 붙여넣기
